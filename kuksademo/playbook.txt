# This is my cheat sheet for an introduction
# to VSS and Eclipse Kuksa.
# Obviously you would not need all commands,
# or need to modify them a bit. "Works for me"
# But you get the idea :)



docker run -it --rm --net=host ghcr.io/eclipse-kuksa/kuksa-databroker:0.5 --enable-viss
# Some Mac OS versions can not use port 55555 as it is reserved by the OS. You can bind it, but can never connect to it.
MAC:
docker run -it --rm --net=host ghcr.io/eclipse-kuksa/kuksa-databroker:0.6 --port 55556 --enable-viss




docker run -it --rm --net=host ghcr.io/eclipse-kuksa/kuksa-python-sdk/kuksa-client:0.5
MAC:
docker run -it --rm --net=host ghcr.io/eclipse-kuksa/kuksa-python-sdk/kuksa-client:0.5 grpc://127.0.0.1:55556

# In the Kuksa client run:

# Should return no value since the Vehicle.Speed has not been set yet.
getValue Vehicle.Speed

# Set Vehicle.Speed and check if it was stored.
setValue Vehicle.Speed 200
getMetaData Vehicle.Speed



# Get the Metadata for another signal
getMetaData Vehicle.Chassis.Axle.Row1.Wheel.Left.Brake.PadWear
# Showcase that the databroker only stores the value for
# a signal that has been set most recently
getValue Vehicle.Chassis.Axle.Row1.Wheel.Left.Brake.PadWear
setValue Vehicle.Chassis.Axle.Row1.Wheel.Left.Brake.PadWear 200
setValue Vehicle.Chassis.Axle.Row1.Wheel.Left.Brake.PadWear 50
getValue Vehicle.Chassis.Axle.Row1.Wheel.Left.Brake.PadWear
